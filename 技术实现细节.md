## 3. 物品比对算法

### 比对规则

```python
def compare(entry_items, exit_items):
    """
    比对入场和离场物品
    
    规则:
    1. 物品类型必须一致
    2. 物品数量必须一致
    3. 如果不一致，标记为异常
    """
    
    # 统计物品
    entry_dict = {}
    for item in entry_items:
        key = item["name"]
        entry_dict[key] = entry_dict.get(key, 0) + item["quantity"]
    
    exit_dict = {}
    for item in exit_items:
        key = item["name"]
        exit_dict[key] = exit_dict.get(key, 0) + item["quantity"]
    
    # 比对
    anomalies = []
    
    # 检查少带物品
    for item_name, quantity in entry_dict.items():
        if item_name not in exit_dict:
            anomalies.append({
                "type": "missing_item",
                "description": f"少带物品：{item_name}（{quantity}个）"
            })
        elif exit_dict[item_name] < quantity:
            anomalies.append({
                "type": "quantity_mismatch",
                "description": f"物品数量不符：{item_name}（入场{quantity}个，离场{exit_dict[item_name]}个）"
            })
    
    # 检查多带物品
    for item_name, quantity in exit_dict.items():
        if item_name not in entry_dict:
            anomalies.append({
                "type": "extra_item",
                "description": f"多带物品：{item_name}（{quantity}个）"
            })
    
    # 判断状态
    status = "anomaly" if anomalies else "normal"
    
    return status, anomalies
```
