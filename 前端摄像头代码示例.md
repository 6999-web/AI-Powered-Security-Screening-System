### 2. 在 `<script setup>` 部分添加摄像头逻辑

```javascript
import { ref, onUnmounted } from 'vue'
import { Camera, MagicStick } from '@element-plus/icons-vue'
import { ElMessage } from 'element-plus'
import api from '@/api'

// 摄像头相关状态
const cameraActive = ref(false)
const videoElement = ref(null)
const mediaStream = ref(null)
const imagePreview = ref(null)
const imageFile = ref(null)

// 识别相关状态
const recognizing = ref(false)
const recognitionProgress = ref(0)
const recognitionStatus = ref('')
const recognitionResult = ref(null)

/**
 * 启动摄像头
 */
const startCamera = async () => {
  try {
    // 请求摄像头权限
    const stream = await navigator.mediaDevices.getUserMedia({
      video: {
        width: { ideal: 1280 },
        height: { ideal: 720 },
        facingMode: 'environment' // 使用后置摄像头
      }
    })
    
    mediaStream.value = stream
    cameraActive.value = true
    
    // 等待 DOM 更新
    await nextTick()
    
    // 将视频流绑定到 video 元素
    if (videoElement.value) {
      videoElement.value.srcObject = stream
    }
    
    ElMessage.success('摄像头已启动')
  } catch (error) {
    console.error('摄像头启动失败:', error)
    ElMessage.error('无法访问摄像头，请检查权限设置')
  }
}

/**
 * 停止摄像头
 */
const stopCamera = () => {
  if (mediaStream.value) {
    mediaStream.value.getTracks().forEach(track => track.stop())
    mediaStream.value = null
  }
  cameraActive.value = false
}

/**
 * 拍照
 */
const capturePhoto = () => {
  if (!videoElement.value) return
  
  // 创建 canvas 元素
  const canvas = document.createElement('canvas')
  canvas.width = videoElement.value.videoWidth
  canvas.height = videoElement.value.videoHeight
  
  // 绘制当前帧
  const ctx = canvas.getContext('2d')
  ctx.drawImage(videoElement.value, 0, 0)
  
  // 转换为 Blob
  canvas.toBlob((blob) => {
    imageFile.value = new File([blob], 'camera-photo.jpg', { type: 'image/jpeg' })
    imagePreview.value = URL.createObjectURL(blob)
    
    // 停止摄像头
    stopCamera()
    
    ElMessage.success('拍照成功')
  }, 'image/jpeg', 0.9)
}

/**
 * 重新拍摄
 */
const retakePhoto = () => {
  imagePreview.value = null
  imageFile.value = null
  startCamera()
}

/**
 * 开始识别
 */
const startRecognition = async () => {
  if (!imageFile.value) {
    ElMessage.warning('请先拍摄照片')
    return
  }
  
  recognizing.value = true
  recognitionProgress.value = 0
  recognitionStatus.value = '正在上传图片...'
  
  try {
    // 模拟进度
    const progressInterval = setInterval(() => {
      if (recognitionProgress.value < 90) {
        recognitionProgress.value += 10
      }
    }, 200)
    
    // 构建表单数据
    const formData = new FormData()
    formData.append('image', imageFile.value)
    formData.append('userId', form.value.userId)
    formData.append('weight', form.value.weight || 0)
    formData.append('channelNo', form.value.channelNo)
    
    recognitionStatus.value = '正在识别物品...'
    
    // 调用 API
    const result = await api.checkIn(formData)
    
    clearInterval(progressInterval)
    recognitionProgress.value = 100
    recognitionStatus.value = '识别完成！'
    
    // 保存识别结果
    recognitionResult.value = result
    
    ElMessage.success(`识别成功！共识别到 ${result.totalCount} 件物品`)
    
    // 跳转到下一步
    setTimeout(() => {
      currentStep.value = 3
    }, 1000)
    
  } catch (error) {
    console.error('识别失败:', error)
    ElMessage.error('识别失败：' + (error.message || '未知错误'))
  } finally {
    recognizing.value = false
  }
}

// 组件卸载时停止摄像头
onUnmounted(() => {
  stopCamera()
})
```

## 完整流程

1. 用户点击"调用摄像头"按钮
2. 浏览器请求摄像头权限
3. 显示实时视频预览
4. 用户点击"拍照"按钮
5. Canvas 截取当前帧并转换为图片
6. 显示图片预览
7. 用户点击"开始识别"
8. 上传图片到后端
9. 后端调用通义千问 API 识别
10. 返回识别结果并展示

## 注意事项

1. 摄像头需要 HTTPS 或 localhost 环境
2. 用户需要授予摄像头权限
3. 建议添加图片压缩功能降低成本
4. 识别时间约 2-3 秒，需要加载提示
